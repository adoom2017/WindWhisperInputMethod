cmake_minimum_required(VERSION 3.16)
project(FengYuCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(fy_engine SHARED src/fy_engine.cpp)
target_compile_definitions(fy_engine PRIVATE FY_ENGINE_BUILD)
target_include_directories(fy_engine PUBLIC include)
add_executable(fy_engine_golden tests/golden.cpp)
target_link_libraries(fy_engine_golden PRIVATE fy_engine)
enable_testing()
add_test(NAME golden COMMAND fy_engine_golden)
